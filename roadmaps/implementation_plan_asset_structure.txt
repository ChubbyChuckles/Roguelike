## 2. JSON Schema Examples

### Sprite Sheet Definition (sprite.schema.json)
```json
{
  "$schema": "http://json-schema.org/draft-07/schema#",
  "title": "Sprite Asset Definition",
  "type": "object",
  "required": ["version", "name", "texture", "sprites"],
  "properties": {
    "version": {
      "type": "string",
      "pattern": "^\\d+\\.\\d+\\.\\d+$"
    },
    "name": {
      "type": "string",
      "description": "Human-readable name of the sprite asset"
    },
    "texture": {
      "type": "object",
      "required": ["file", "width", "height"],
      "properties": {
        "file": {
          "type": "string",
          "description": "Relative path to texture file"
        },
        "width": {"type": "integer", "minimum": 1},
        "height": {"type": "integer", "minimum": 1},
        "format": {
          "type": "string",
          "enum": ["PNG", "BMP", "JPG"],
          "default": "PNG"
        }
      }
    },
    "sprites": {
      "type": "array",
      "items": {
        "type": "object",
        "required": ["id", "x", "y", "width", "height"],
        "properties": {
          "id": {"type": "string"},
          "x": {"type": "integer", "minimum": 0},
          "y": {"type": "integer", "minimum": 0},
          "width": {"type": "integer", "minimum": 1},
          "height": {"type": "integer", "minimum": 1},
          "pivot_x": {"type": "number", "default": 0.5},
          "pivot_y": {"type": "number", "default": 0.5}
        }
      }
    },
    "animations": {
      "type": "array",
      "items": {
        "type": "object",
        "required": ["name", "frames"],
        "properties": {
          "name": {"type": "string"},
          "frames": {
            "type": "array",
            "items": {
              "type": "object",
              "required": ["sprite_id", "duration"],
              "properties": {
                "sprite_id": {"type": "string"},
                "duration": {"type": "number", "minimum": 0}
              }
            }
          },
          "loop": {"type": "boolean", "default": true}
        }
      }
    }
  }
}
```

## 6. GitHub Actions Integration

### Extended CI Workflow (asset-validation.yml)
```yaml
name: Asset Validation and Testing

on:
  push:
    paths:
      - 'assets/**'
      - 'src/asset_manager/**'
  pull_request:
    paths:
      - 'assets/**'
      - 'src/asset_manager/**'

jobs:
  validate-assets:
    runs-on: ${{ matrix.os }}
    strategy:
      matrix:
        os: [windows-latest, ubuntu-latest, macos-latest]

    steps:
    - uses: actions/checkout@v3

    - name: Setup Python for validation scripts
      uses: actions/setup-python@v4
      with:
        python-version: '3.9'

    - name: Install Python dependencies
      run: |
        pip install jsonschema pillow

    - name: Validate JSON schemas
      run: python scripts/validate_assets.py --schema-dir assets/data/schemas --asset-dir assets

    - name: Check asset file integrity
      run: python scripts/check_asset_integrity.py --asset-dir assets

    - name: Validate image formats
      run: python scripts/validate_images.py --image-dir assets/graphics

    - name: Generate asset manifest
      run: python scripts/generate_manifest.py --output-dir build/manifests

    - name: Upload validation results
      uses: actions/upload-artifact@v3
      if: always()
      with:
        name: asset-validation-${{ matrix.os }}
        path: build/validation-reports/
```

### Python Validation Script (validate_assets.py)
```python
#!/usr/bin/env python3
import json
import jsonschema
import os
import sys
from pathlib import Path

def validate_json_file(json_path, schema_path):
    """Validate a JSON file against its schema."""
    try:
        with open(json_path, 'r') as f:
            data = json.load(f)

        with open(schema_path, 'r') as f:
            schema = json.load(f)

        jsonschema.validate(data, schema)
        return True, None
    except jsonschema.exceptions.ValidationError as e:
        return False, str(e)
    except Exception as e:
        return False, f"Error processing file: {str(e)}"
```
