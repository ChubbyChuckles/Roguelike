# Backfill critical compile definitions for tests that may have been generated before
# SDL_MAIN_HANDLED and TEST_COMBAT_PERMISSIVE were added in this file.
foreach(tgt IN LISTS UNIT_TESTS)
    get_filename_component(_tname ${tgt} NAME_WE)
    if(TARGET ${_tname})
        # Prevent SDL from hijacking main() and enable permissive combat checks in tests
        target_compile_definitions(${_tname} PRIVATE SDL_MAIN_HANDLED=1 TEST_COMBAT_PERMISSIVE=1)
        if(ROGUE_ENABLE_SDL)
            target_compile_definitions(${_tname} PRIVATE ROGUE_HAVE_SDL=1)
        endif()
    endif()
endforeach()

    if(DEFINED ROGUE_SDL2_DLL_DIR AND NOT ROGUE_SDL2_DLL_DIR STREQUAL "")
        set(_SDL2_DLL_PATH "${ROGUE_SDL2_DLL_DIR}/SDL2.dll")
        if(EXISTS "${_SDL2_DLL_PATH}")
            get_property(_tests_targets DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY BUILDSYSTEM_TARGETS)
            foreach(_tg IN LISTS _tests_targets)
                if(_tg MATCHES "^test_")
                    add_custom_command(TARGET ${_tg} POST_BUILD

                        COMMENT "Copy SDL2.dll to test output for ${_tg}")
                endif()
            endforeach()
        endif()
    endif()
endif()
