foreach(ut ${UNIT_TESTS})
    get_filename_component(name ${ut} NAME_WE)
    if(TARGET ${name})
        # Skip if already added explicitly above
        continue()
    endif()
    add_executable(${name} ${ut})
    target_link_libraries(${name} PRIVATE rogue_core)
    # Prevent SDL from redefining main to SDL_main when SDL headers transitively included
    target_compile_definitions(${name} PRIVATE SDL_MAIN_HANDLED=1)
    target_compile_definitions(${name} PRIVATE TEST_COMBAT_PERMISSIVE=1)
    if(ROGUE_ENABLE_SDL)
        target_compile_definitions(${name} PRIVATE ROGUE_HAVE_SDL=1)
    endif()
    add_test(NAME ${name} COMMAND ${name})
    # Ensure SDL runtime is discoverable in CTest on Windows; prefer directory provided from parent scope
    if(WIN32 AND ROGUE_ENABLE_SDL)
        # Headless renderer and PATH tweak
        set_property(TEST ${name} APPEND PROPERTY ENVIRONMENT "SDL_VIDEODRIVER=dummy")
        if(DEFINED ROGUE_SDL2_DLL_DIR AND NOT ROGUE_SDL2_DLL_DIR STREQUAL "")
            set_property(TEST ${name} APPEND PROPERTY ENVIRONMENT "PATH=${ROGUE_SDL2_DLL_DIR};$ENV{PATH}")
        endif()
    endif()
endforeach()

foreach(it ${INTEGRATION_TESTS})
    get_filename_component(name ${it} NAME_WE)
    add_executable(${name} ${it})
    target_link_libraries(${name} PRIVATE rogue_core)
    if(ROGUE_ENABLE_SDL)
        target_compile_definitions(${name} PRIVATE ROGUE_HAVE_SDL=1)
    endif()
    if(ROGUE_ENABLE_SDL)
        add_test(NAME ${name} COMMAND ${name})
        # Headless renderer for CI and PATH fix on Windows
        if(WIN32)
            set_property(TEST ${name} APPEND PROPERTY ENVIRONMENT "SDL_VIDEODRIVER=dummy")
            if(DEFINED ROGUE_SDL2_DLL_DIR AND NOT ROGUE_SDL2_DLL_DIR STREQUAL "")
                set_property(TEST ${name} APPEND PROPERTY ENVIRONMENT "PATH=${ROGUE_SDL2_DLL_DIR};$ENV{PATH}")
            endif()
        else()
            set_property(TEST ${name} APPEND PROPERTY ENVIRONMENT "SDL_VIDEODRIVER=dummy")
        endif()
    else()
        add_test(NAME ${name} COMMAND ${name})
    endif()
endforeach()

# Backfill critical compile definitions for tests that may have been generated before
# SDL_MAIN_HANDLED and TEST_COMBAT_PERMISSIVE were added in this file.
foreach(tgt IN LISTS UNIT_TESTS)
    get_filename_component(_tname ${tgt} NAME_WE)
    if(TARGET ${_tname})
        # Prevent SDL from hijacking main() and enable permissive combat checks in tests
        target_compile_definitions(${_tname} PRIVATE SDL_MAIN_HANDLED=1 TEST_COMBAT_PERMISSIVE=1)
        if(ROGUE_ENABLE_SDL)
            target_compile_definitions(${_tname} PRIVATE ROGUE_HAVE_SDL=1)
        endif()
    endif()
endforeach()

# After all tests are declared, copy SDL2.dll next to test executables on Windows
if(WIN32 AND ROGUE_ENABLE_SDL)
    if(DEFINED ROGUE_SDL2_DLL_DIR AND NOT ROGUE_SDL2_DLL_DIR STREQUAL "")
        set(_SDL2_DLL_PATH "${ROGUE_SDL2_DLL_DIR}/SDL2.dll")
        if(EXISTS "${_SDL2_DLL_PATH}")
            get_property(_tests_targets DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY BUILDSYSTEM_TARGETS)
            foreach(_tg IN LISTS _tests_targets)
                if(_tg MATCHES "^test_")
                    add_custom_command(TARGET ${_tg} POST_BUILD
                        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_SDL2_DLL_PATH}" $<TARGET_FILE_DIR:${_tg}>
                        COMMENT "Copy SDL2.dll to test output for ${_tg}")
                endif()
            endforeach()
        endif()
    endif()
endif()
